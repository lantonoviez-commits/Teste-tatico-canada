// Função matemática idêntica à do site da prova para validar o código function verificarAssinaturaProva(codigoCompleto) { const partes = codigoCompleto.trim().toUpperCase().split("-"); if (partes.length !== 4) return null; const [idPass, tentativa, acertosStr, hashRecebido] = partes; const acertos = parseInt(acertosStr, 10); if (isNaN(acertos)) return null; const raw = `RES_${idPass}_${tentativa}_${acertos}_CANADA_AREA_38_OFFICIAL_HASH`; let hash = 0; for (let i = 0; i < raw.length; i++) { hash = (hash << 5) - hash + raw.charCodeAt(i); hash |= 0; } const hashEsperado = Math.abs(hash).toString(16).toUpperCase(); if (hashRecebido !== hashEsperado) return null; return { idPass, tentativa, acertos }; } if (interaction.isButton() && interaction.customId === "btn_tatico_validar_chave") { const modal = new ModalBuilder() .setCustomId("modal_validar_prova_tatica") .setTitle("Validar Código da Prova Teórica"); modal.addComponents( new ActionRowBuilder().addComponents( new TextInputBuilder() .setCustomId("prova_nome_id") .setLabel("SEU NOME IN-GAME") .setPlaceholder("Ex: Saturo Yokai") .setStyle(TextInputStyle.Short) .setRequired(true) ), new ActionRowBuilder().addComponents( new TextInputBuilder() .setCustomId("prova_codigo_validacao") .setLabel("CÓDIGO DE VALIDAÇÃO GERADO NO FIM DA PROVA") .setPlaceholder("Ex: 9650-1-8-4A9F12C0") .setStyle(TextInputStyle.Short) .setRequired(true) ) ); return interaction.showModal(modal); } if (interaction.isModalSubmit() && interaction.customId === "modal_validar_prova_tatica") { const nomeInGame = interaction.fields.getTextInputValue("prova_nome_id").trim(); const codigoDigitado = interaction.fields.getTextInputValue("prova_codigo_validacao").trim(); const resultadoValido = verificarAssinaturaProva(codigoDigitado); if (!resultadoValido) { return interaction.reply({ content: "❌ **Código de validação inválido ou adulterado!** Copie exatamente o código exibido na tela final da prova.", ephemeral: true }); } const { idPass, tentativa, acertos } = resultadoValido; const aprovado = acertos >= CONFIG.NOTA_MINIMA_PROVA; if (!aprovado) { return interaction.reply({ content: `❌ Esse código corresponde a uma nota de **${acertos}/10** (mínima necessária: **${CONFIG.NOTA_MINIMA_PROVA}/10**). Você precisa ser aprovado para liberar o teste prático.`, ephemeral: true }); } const taticoData = carregarTatico(); taticoData.provasAprovadas[interaction.user.id] = { nomeId: `${nomeInGame} | ${idPass}`, nota: acertos, tentativa, chave: codigoDigitado.toUpperCase(), data: Date.now() }; salvarTatico(taticoData); return interaction.reply({ content: `✅ **Prova Teórica validada com sucesso!** (Nota oficial: **${acertos}/10** | Passaporte: \`${idPass}\`).\nAgora você já pode clicar em **Solicitar Teste Prático** (disponível a partir das **18:00**).`, ephemeral: true }); }
